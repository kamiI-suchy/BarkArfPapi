-- Uwagi:
--   - literały dat: {d 'YYYY-MM-DD'}

-- ============================================================
-- ZADANIE 1: Utworzenie tabel (bez pól Premia i Specjalizacja)
-- Kolejność: PRACOWNICY → DZIALY → ZESPOLY → FK PRACOWNICY→ZESPOLY
-- bo DZIALY i ZESPOLY mają FK do PRACOWNICY
-- a PRACOWNICY ma FK do ZESPOLY (cykliczne - dodane ALTER TABLE)
-- ============================================================

CREATE TABLE PRACOWNICY (
    NR_PRAC INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    IMIE VARCHAR(50),
    NAZWISKO VARCHAR(50),
    STANOWISKO VARCHAR(100),
    ZAROBKI DECIMAL(10,2),
    DATA_ZATRUDNIENIA DATE,
    NR_ZESPOLU INTEGER
);

CREATE TABLE DZIALY (
    NR_DZIALU INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NAZWA VARCHAR(100),
    LOKALIZACJA VARCHAR(100),
    SZEF_DZIALU INTEGER,
    FOREIGN KEY (SZEF_DZIALU) REFERENCES PRACOWNICY(NR_PRAC)
);

CREATE TABLE ZESPOLY (
    NR_ZESPOLU INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NAZWA_ZESPOLU VARCHAR(100),
    DATA_POWOLANIA DATE,
    BUDZET DECIMAL(10,2),
    KIEROWNIK INTEGER,
    NR_DZIALU INTEGER,
    FOREIGN KEY (KIEROWNIK) REFERENCES PRACOWNICY(NR_PRAC),
    FOREIGN KEY (NR_DZIALU) REFERENCES DZIALY(NR_DZIALU)
);

ALTER TABLE PRACOWNICY ADD FOREIGN KEY (NR_ZESPOLU) REFERENCES ZESPOLY(NR_ZESPOLU);


-- ============================================================
-- ZADANIE 2: Dodanie pól Premia i Specjalizacja
-- ============================================================

ALTER TABLE PRACOWNICY ADD COLUMN PREMIA DECIMAL(10,2);

ALTER TABLE ZESPOLY ADD COLUMN SPECJALIZACJA VARCHAR(100);

-- ============================================================
-- ZADANIE 6: Imiona i nazwiska pracowników zarabiających >= 4489.34 zł
-- ============================================================

SELECT IMIE, NAZWISKO
FROM PRACOWNICY
WHERE ZAROBKI >= 4489.34;


-- ============================================================
-- ZADANIE 7: Lista pracowników wraz z nazwami zespołów
-- ============================================================

SELECT P.IMIE, P.NAZWISKO, Z.NAZWA_ZESPOLU
FROM PRACOWNICY P
LEFT JOIN ZESPOLY Z ON P.NR_ZESPOLU = Z.NR_ZESPOLU;


-- ============================================================
-- ZADANIE 8: Lista działów wraz z nazwiskami ich szefów
-- ============================================================

SELECT D.NAZWA, P.NAZWISKO
FROM DZIALY D
JOIN PRACOWNICY P ON D.SZEF_DZIALU = P.NR_PRAC;


-- ============================================================
-- ZADANIE 9: Działy, które nie mają przyporządkowanego żadnego zespołu
-- ============================================================

SELECT NAZWA
FROM DZIALY
WHERE NR_DZIALU NOT IN (
    SELECT DISTINCT NR_DZIALU
    FROM ZESPOLY
    WHERE NR_DZIALU IS NOT NULL
);


-- ============================================================
-- ZADANIE 10: Pracownicy nie będący szefem żadnego działu
-- ============================================================

SELECT NAZWISKO
FROM PRACOWNICY
WHERE NR_PRAC NOT IN (
    SELECT SZEF_DZIALU
    FROM DZIALY
    WHERE SZEF_DZIALU IS NOT NULL
);


-- ============================================================
-- ZADANIE 11: Kierownicy zespołów zatrudnieni po roku 2001 (X1)
--              Wypisać nazwiska i zarobki
-- ============================================================

SELECT P.NAZWISKO, P.ZAROBKI
FROM PRACOWNICY P
JOIN ZESPOLY Z ON P.NR_PRAC = Z.KIEROWNIK
WHERE EXTRACT(YEAR FROM P.DATA_ZATRUDNIENIA) > 2001;


-- ============================================================
-- ZADANIE 12: Osoby będące jednocześnie kierownikiem zespołu
--              oraz szefem działu
-- ============================================================

SELECT P.NAZWISKO
FROM PRACOWNICY P
WHERE P.NR_PRAC IN (SELECT KIEROWNIK FROM ZESPOLY)
  AND P.NR_PRAC IN (SELECT SZEF_DZIALU FROM DZIALY);


-- ============================================================
-- ZADANIE 13: Osoby z zespołów o budżecie > 16000 (X2)
-- ============================================================

SELECT P.IMIE, P.NAZWISKO
FROM PRACOWNICY P
JOIN ZESPOLY Z ON P.NR_ZESPOLU = Z.NR_ZESPOLU
WHERE Z.BUDZET > 16000;


-- ============================================================
-- ZADANIE 14: Osoby zatrudnione zanim powstał ich zespół
--              (DATA_ZATRUDNIENIA < DATA_POWOLANIA)
-- ============================================================

SELECT P.IMIE, P.NAZWISKO, Z.NAZWA_ZESPOLU
FROM PRACOWNICY P
JOIN ZESPOLY Z ON P.NR_ZESPOLU = Z.NR_ZESPOLU
WHERE P.DATA_ZATRUDNIENIA < Z.DATA_POWOLANIA;


-- ============================================================
-- ZADANIE 15: Osoby na stanowisku "programista" (X3)
--              w zespołach z budżetem <= 3000 (X4)
-- ============================================================

SELECT P.IMIE, P.NAZWISKO
FROM PRACOWNICY P
JOIN ZESPOLY Z ON P.NR_ZESPOLU = Z.NR_ZESPOLU
WHERE P.STANOWISKO = 'programista'
  AND Z.BUDZET <= 3000;
